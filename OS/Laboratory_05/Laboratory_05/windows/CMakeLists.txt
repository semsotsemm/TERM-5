cmake_minimum_required(VERSION 3.16)
project(Laboratory_05 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Threads REQUIRED)
add_executable(Lab-05a Lab-05a/Lab-05a.cpp)
target_link_libraries(Lab-05a PRIVATE Threads::Threads)
add_executable(Lab-05x Lab-05x/Lab-05x.cpp)
target_link_libraries(Lab-05x PRIVATE Threads::Threads)
add_executable(Lab-05b Lab-05b/Lab-05b.cpp)
target_link_libraries(Lab-05b PRIVATE Threads::Threads)
add_executable(Lab-05c Lab-05c/Lab-05c.cpp)
target_link_libraries(Lab-05c PRIVATE Threads::Threads)
add_executable(Lab-05SJF Lab-05SJF/Lab-05SJF.cpp common/Scheduler.cpp)
target_link_libraries(Lab-05SJF PRIVATE Threads::Threads)
add_executable(Lab-05RR Lab-05RR/Lab-05RR.cpp common/Scheduler.cpp)
target_link_libraries(Lab-05RR PRIVATE Threads::Threads)
add_executable(Lab-05CP Lab-05CP/Lab-05CP.cpp common/Scheduler.cpp)
target_link_libraries(Lab-05CP PRIVATE Threads::Threads)
add_executable(Lab-05MQ Lab-05MQ/Lab-05MQ.cpp common/Scheduler.cpp)
target_link_libraries(Lab-05MQ PRIVATE Threads::Threads)
if(UNIX AND NOT APPLE)
  foreach(target Lab-05a Lab-05x Lab-05b Lab-05c)
    target_link_libraries(${target} PRIVATE rt)
  endforeach()
endif()
if(MSVC)
  foreach(target Lab-05a Lab-05x Lab-05b Lab-05c Lab-05SJF Lab-05RR Lab-05CP Lab-05MQ)
    target_compile_options(${target} PRIVATE /utf-8)
  endforeach()
endif()
configure_file(tasks.csv "${CMAKE_BINARY_DIR}/tasks.csv" COPYONLY)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
configure_file(tasks.csv "${CMAKE_BINARY_DIR}/bin/tasks.csv" COPYONLY)
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${config}")
  configure_file(tasks.csv "${CMAKE_BINARY_DIR}/bin/${config}/tasks.csv" COPYONLY)
endforeach()
add_dependencies(Lab-05b Lab-05x)
